1. Bài toán 1: Priority Async Scheduler (Giới hạn đồng thời & Độ ưu tiên)

Yêu cầu phỏng vấn: Thiết kế class PriorityScheduler cho phép nhận các tác vụ bất đồng bộ (async tasks), chỉ cho phép tối đa concurrency tác vụ chạy cùng lúc, và tự động ưu tiên tác vụ có priority cao hơn khi có slot trống.

JavaScript — Cài đặt chuẩn Senior 
export class PriorityScheduler {
  constructor(concurrency = 2) {
    this.concurrency = concurrency;
    this.runningCount = 0;
    this.queue = [];
  }

  add(task, priority = 0) {
    return new Promise((resolve, reject) => {
      this.queue.push({ task, priority, resolve, reject });
      this.queue.sort((a, b) => b.priority - a.priority); // Ưu tiên cao nhất đứng đầu
      this._runNext();
    });
  }

  async _runNext() {
    if (this.runningCount >= this.concurrency || this.queue.length === 0) return;

    const { task, resolve, reject } = this.queue.shift();
    this.runningCount++;

    try {
      const result = await task();
      resolve(result);
    } catch (err) {
      reject(err);
    } finally {
      this.runningCount--;
      this._runNext(); // Kích hoạt task tiếp theo trong hàng đợi
    }
  }
}

2. Bài toán 2: Deep Object Diff & Patch Engine (State Reconciler)

Yêu cầu phỏng vấn: Viết hàm deepDiff(oldObj, newObj) trả về sự khác biệt giữa 2 object lồng nhau { added, updated, deleted } và hàm applyPatch(target, patch) để khôi phục lại state mới.

JavaScript — Deep Object Diff & Patch Implementation 
export function deepDiff(oldObj, newObj) {
  const diff = { added: {}, updated: {}, deleted: [] };
  const isObj = val => val !== null && typeof val === 'object' && !Array.isArray(val);

  for (const key of Object.keys(oldObj)) {
    if (!(key in newObj)) {
      diff.deleted.push(key);
    } else if (isObj(oldObj[key]) && isObj(newObj[key])) {
      const nested = deepDiff(oldObj[key], newObj[key]);
      if (Object.keys(nested.added).length || Object.keys(nested.updated).length || nested.deleted.length) {
        diff.updated[key] = nested;
      }
    } else if (oldObj[key] !== newObj[key]) {
      diff.updated[key] = { from: oldObj[key], to: newObj[key] };
    }
  }

  for (const key of Object.keys(newObj)) {
    if (!(key in oldObj)) diff.added[key] = newObj[key];
  }
  return diff;
}

3. Bài toán 3: Mini Signal System (Fine-grained Reactivity & Dependency Tracking)

Yêu cầu phỏng vấn: Xây dựng cơ chế Reactive Signal (nền tảng của SolidJS, Vue 3 Reactivity, Preact Signals, Angular Signals) với createSignal và createEffect có khả năng tự động thu thập phụ thuộc (Dependency Collection).

JavaScript — Fine-Grained Reactive Signal Architecture 
let activeEffect = null;

export function createSignal(initialValue) {
  let value = initialValue;
  const subscribers = new Set();

  function read() {
    if (activeEffect) subscribers.add(activeEffect); // Thu thập dependency
    return value;
  }

  function write(newValue) {
    const next = typeof newValue === 'function' ? newValue(value) : newValue;
    if (next !== value) {
      value = next;
      new Set(subscribers).forEach(fn => fn()); // Thông báo toàn bộ subscribers
    }
  }

  return [read, write];
}

export function createEffect(fn) {
  const effectRunner = () => {
    activeEffect = effectRunner;
    try { fn(); } finally { activeEffect = null; }
  };
  effectRunner(); // Chạy lần đầu để thiết lập liên kết
}
