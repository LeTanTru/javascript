4. AbortController Pattern (Cancel-on-Supersede)

JavaScript — Cẩm nang Cancel-on-Supersede 
let activeController = null;

async function search(query) {
  activeController?.abort('superseded'); // Hủy request cũ
  activeController = new AbortController();

  try {
    const res = await fetch(`/api?q=${query}`, { signal: activeController.signal });
    renderResults(await res.json());
  } catch (e) {
    if (e.name !== 'AbortError') throw e; // Chỉ bỏ qua AbortError
  }
}

5. Concurrency Limiter (p-limit Pattern)

JavaScript — Cẩm nang Concurrency Limiter + asyncPool 
function createConcurrencyLimiter(concurrency) {
  const queue = []; let activeCount = 0;

  const next = () => {
    if (!queue.length || activeCount >= concurrency) return;
    activeCount++;
    const { taskFactory, resolve, reject } = queue.shift();
    Promise.resolve(taskFactory()).then(resolve, reject).finally(() => {
      activeCount--; next();
    });
  };

  return (taskFactory) => new Promise((resolve, reject) => {
    queue.push({ taskFactory, resolve, reject }); next();
  });
}

// asyncPool: Batch processing với concurrency control
const asyncPool = (items, n, fn) => {
  const limit = createConcurrencyLimiter(n);
  return Promise.all(items.map((item) => limit(() => fn(item))));
};
