1. Kiến trúc Pub/Sub & Live-coding EventEmitter

EventEmitter là hiện thực của mẫu thiết kế Publish/Subscribe (Observer Pattern). Một thành phần phát ra sự kiện mà không cần biết ai đang lắng nghe nó, giúp các module độc lập tuyệt đối.

JavaScript — Cài đặt EventEmitter chuẩn Node.js 
class EventEmitter {
  constructor() {
    this.events = new Map(); // Map: eventName -> Array of listeners
  }

  // Đăng ký sự kiện
  on(eventName, listener) {
    if (typeof listener !== 'function') throw new TypeError('Listener must be a function');
    if (!this.events.has(eventName)) {
      this.events.set(eventName, []);
    }
    this.events.get(eventName).push(listener);

    // Trả về Subscription object tiện lợi cho cleanup
    return {
      unsubscribe: () => this.off(eventName, listener),
    };
  }

  // Phát sự kiện kèm dữ liệu
  emit(eventName, ...args) {
    if (!this.events.has(eventName)) return [];
    // Copy mảng listeners để tránh lỗi khi listener tự gỡ bỏ chính nó trong lúc emit
    const listeners = [...this.events.get(eventName)];
    return listeners.map(listener => listener.apply(this, args));
  }

  // Hủy đăng ký listener
  off(eventName, listener) {
    if (!this.events.has(eventName)) return this;
    const listeners = this.events.get(eventName);
    const index = listeners.indexOf(listener);
    if (index !== -1) listeners.splice(index, 1);
    if (listeners.length === 0) this.events.delete(eventName);
    return this;
  }

  // Lắng nghe đúng 1 lần rồi tự hủy
  once(eventName, listener) {
    const wrapper = (...args) => {
      this.off(eventName, wrapper);
      return listener.apply(this, args);
    };
    return this.on(eventName, wrapper);
  }
}

2. Function Composition: pipe vs compose

Thay vì viết các hàm lồng nhau khó đọc h(g(f(x))), lập trình hàm sử dụng Composition để xâu chuỗi các hàm thuần khiết thành một pipeline duy nhất :

Kỹ thuậtHướng thực thiCông thức toán họcGhi chú
pipe(f, g, h) Trái sang Phải (Left-to-Right) $h(g(f(x)))$ Tự nhiên theo dòng đọc mã nguồn của con người.
compose(f, g, h) Phải sang Trái (Right-to-Left) $f(g(h(x)))$ Chuẩn ký hiệu hàm hợp toán học $(f \circ g \circ h)(x)$ (Redux middleware).
JavaScript — Cài đặt pipe và compose bằng Array.prototype.reduce 
// pipe: Trái -> Phải (Reduce từ đầu đến cuối)
function pipe(...fns) {
  return function (...initialArgs) {
    if (fns.length === 0) return initialArgs[0];
    const [firstFn, ...restFns] = fns;
    return restFns.reduce((acc, fn) => fn(acc), firstFn(...initialArgs));
  };
}

// compose: Phải -> Trái (Reduce từ cuối về đầu)
function compose(...fns) {
  return function (...initialArgs) {
    if (fns.length === 0) return initialArgs[0];
    const reversed = [...fns].reverse();
    const [firstFn, ...restFns] = reversed;
    return restFns.reduce((acc, fn) => fn(acc), firstFn(...initialArgs));
  };
}

3. Mở rộng: Async Pipeline (pipeAsync)

JavaScript — pipeAsync hỗ trợ Promise 
function pipeAsync(...fns) {
  return function (...initialArgs) {
    return fns.reduce(async (accPromise, fn) => {
      const acc = await accPromise;
      return fn(acc);
    }, Promise.resolve(fns.length > 0 ? fns[0](...initialArgs) : initialArgs[0]));
  };
}

// Sử dụng: Pipeline xác thực & xử lý user
const processOrder = pipeAsync(
  validateCart,     // async (cart) => validCart
  calculateTotal,   // (validCart) => total
  chargeCreditCard, // async (total) => receipt
  sendConfirmation  // async (receipt) => emailStatus
);
