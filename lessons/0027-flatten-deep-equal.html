1. Live-coding: flattenArray() theo độ sâu (Depth)

JavaScript — Array.prototype.flat() polyfill 
function flattenArray(arr, depth = 1) {
  if (!Array.isArray(arr)) throw new TypeError('Expected an array');
  if (depth <= 0) return arr.slice();

  return arr.reduce((acc, item) => {
    if (Array.isArray(item)) {
      acc.push(...flattenArray(item, depth - 1));
    } else {
      acc.push(item);
    }
    return acc;
  }, []);
}

// Sử dụng:
flattenArray([1, [2, [3, [4]]]], 2);        // [1, 2, 3, [4]]
flattenArray([1, [2, [3, [4]]]], Infinity); // [1, 2, 3, 4]

2. Live-coding: flattenObject() (Dot Notation)

Chuyển đổi một object phân cấp lồng nhau thành một object phẳng với các keys nối bằng dấu chấm (thường dùng trong Form state, i18n localization keys hoặc Query URL serialization) :

JavaScript — flattenObject polyfill 
function flattenObject(obj, prefix = '', delimiter = '.') {
  if (obj === null || typeof obj !== 'object' || obj instanceof Date || obj instanceof RegExp) {
    return { [prefix]: obj };
  }

  const result = {};
  const keys = Object.keys(obj);

  if (keys.length === 0 && prefix) {
    result[prefix] = Array.isArray(obj) ? [] : {};
    return result;
  }

  for (const key of keys) {
    const val = obj[key];
    const newKey = prefix ? `${prefix}${delimiter}${key}` : key;

    if (val !== null && typeof val === 'object' && !(val instanceof Date) && !(val instanceof RegExp)) {
      Object.assign(result, flattenObject(val, newKey, delimiter));
    } else {
      result[newKey] = val;
    }
  }

  return result;
}

3. Live-coding: deepEqual() Toàn diện

JavaScript — deepEqual polyfill chuẩn Lodash isEqual 
function deepEqual(a, b) {
  // 1. So sánh cơ bản (Xử lý cả NaN === NaN qua Object.is)
  if (Object.is(a, b)) return true;

  // 2. Nếu 1 trong 2 là null hoặc không phải object
  if (a === null || typeof a !== 'object' || b === null || typeof b !== 'object') {
    return false;
  }

  // 3. Khác constructor (VD: Array vs Object, Date vs Object)
  if (a.constructor !== b.constructor) return false;

  // 4. Xử lý Date & RegExp
  if (a instanceof Date) return a.getTime() === b.getTime();
  if (a instanceof RegExp) return a.source === b.source && a.flags === b.flags;

  // 5. So sánh số lượng và giá trị keys (kể cả Symbol)
  const keysA = Reflect.ownKeys(a);
  const keysB = Reflect.ownKeys(b);

  if (keysA.length !== keysB.length) return false;

  for (const key of keysA) {
    if (!Object.prototype.hasOwnProperty.call(b, key) || !deepEqual(a[key], b[key])) {
      return false;
    }
  }

  return true;
}
