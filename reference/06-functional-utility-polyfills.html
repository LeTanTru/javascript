Tổng hợp kiến thức Module 06 : Functional Programming & Utility Polyfills

Bản đồ tư duy cô đọng toàn bộ 10 bài toán Live-coding & Utility Polyfills kinh điển trong mọi vòng phỏng vấn kỹ thuật từ Bài 22 đến Bài 27.

⏳ ~20 phút · 🎯 Mục tiêu: Nắm vững trọn bộ mã nguồn chuẩn Senior của Curry, Debounce, Throttle, deepClone, EventEmitter, pipe/compose, LRUCache, flatten và deepEqual.

1. Ma trận so sánh các cặp kỹ thuật song hành

Currying vs Partial Application

Currying : $f(a, b, c) \rightarrow f(a)(b)(c)$ (Biến đổi thành chuỗi hàm unary 1 tham số đến khi đủ arity fn.length).

Partial Application : Cố định $K$ tham số trước, trả về hàm nhận toàn bộ $N - K$ tham số còn lại trong 1 lần gọi tiếp theo (tương tự bind).

Debounce vs Throttle

Debounce : Reset timer liên tục, gom chuỗi sự kiện thành 1 lần gọi sau khi người dùng ngừng thao tác (Typeahead, Resize, Auto-save).

Throttle : Nhịp cố định, chỉ thực thi tối đa 1 lần trong mỗi chu kỳ wait ms (Scroll, Drag, Button Click chống spam).

deepClone vs structuredClone

deepClone : Xử lý Circular References qua WeakMap, giữ nguyên Symbol keys và Function/Methods.

structuredClone : API trình duyệt tích hợp sẵn, nhưng ném DataCloneError nếu chứa Function hoặc DOM Nodes.

pipe vs compose

pipe(f, g, h) : Trái sang Phải (Left-to-Right: $h(g(f(x)))$) theo chiều đọc mã nguồn tự nhiên.

compose(f, g, h) : Phải sang Trái (Right-to-Left: $f(g(h(x)))$) theo chuẩn hàm hợp Toán học.

2. Bộ mã nguồn 7 Utility Polyfills mẫu (Senior Cheat-Sheet)

1. curry() — Tự động tích lũy tham số & bảo toàn this 
function curry(fn) {
  return function curried(...args) {
    if (args.length >= fn.length) return fn.apply(this, args);
    return function (...nextArgs) {
      return curried.apply(this, args.concat(nextArgs));
    }.bind(this);
  };
}
2. debounce() — Hỗ trợ Leading, Trailing, .cancel() 
function debounce(fn, wait, { leading = false, trailing = true } = {}) {
  let timerId = null, lastArgs = null, lastThis = null, result;
  const invoke = () => {
    if (lastArgs) { result = fn.apply(lastThis, lastArgs); lastArgs = lastThis = null; }
    return result;
  };
  const debounced = function(...args) {
    lastArgs = args; lastThis = this;
    const isLeading = leading && timerId === null;
    if (timerId) clearTimeout(timerId);
    timerId = setTimeout(() => { timerId = null; if (trailing && !isLeading) invoke(); }, wait);
    if (isLeading) invoke();
    return result;
  };
  debounced.cancel = () => { if (timerId) clearTimeout(timerId); timerId = lastArgs = lastThis = null; };
  return debounced;
}
3. throttle() — Kết hợp Timestamp & Timer 
function throttle(fn, wait, { leading = true, trailing = true } = {}) {
  let timerId = null, lastArgs = null, lastThis = null, lastCallTime = 0, result;
  const invoke = (time) => {
    lastCallTime = time;
    if (lastArgs) { result = fn.apply(lastThis, lastArgs); lastArgs = lastThis = null; }
    return result;
  };
  return function(...args) {
    const now = Date.now();
    lastArgs = args; lastThis = this;
    if (lastCallTime === 0 && !leading) lastCallTime = now;
    const rem = wait - (now - lastCallTime);
    if (rem <= 0 || rem > wait) {
      if (timerId) { clearTimeout(timerId); timerId = null; }
      invoke(now);
    } else if (!timerId && trailing) {
      timerId = setTimeout(() => { timerId = null; lastCallTime = leading ? Date.now() : 0; invoke(Date.now()); }, rem);
    }
    return result;
  };
}
4. deepClone() — WeakMap, Date, RegExp, Map, Set & Symbols 
function deepClone(val, hash = new WeakMap()) {
  if (val === null || typeof val !== 'object') return val;
  if (val instanceof Date) return new Date(val.getTime());
  if (val instanceof RegExp) return new RegExp(val.source, val.flags);
  if (hash.has(val)) return hash.get(val);
  if (val instanceof Map) {
    const m = new Map(); hash.set(val, m);
    val.forEach((v, k) => m.set(deepClone(k, hash), deepClone(v, hash)));
    return m;
  }
  if (val instanceof Set) {
    const s = new Set(); hash.set(val, s);
    val.forEach((v) => s.add(deepClone(v, hash)));
    return s;
  }
  const res = Array.isArray(val) ? [] : Object.create(Object.getPrototypeOf(val));
  hash.set(val, res);
  for (const k of Reflect.ownKeys(val)) {
    const desc = Object.getOwnPropertyDescriptor(val, k);
    if (desc && desc.enumerable) res[k] = deepClone(val[k], hash);
  }
  return res;
}
5. LRUCache — O(1) qua Map Insertion-Order 
class LRUCache {
  constructor(capacity) { this.cap = capacity; this.map = new Map(); }
  get(key) {
    if (!this.map.has(key)) return undefined;
    const val = this.map.get(key);
    this.map.delete(key); this.map.set(key, val); // Đưa lên MRU
    return val;
  }
  put(key, val) {
    if (this.map.has(key)) this.map.delete(key);
    else if (this.map.size >= this.cap) this.map.delete(this.map.keys().next().value);
    this.map.set(key, val);
  }
}
6. deepEqual() — Toàn diện Primitives, Objects, Arrays, Dates, Symbols 
function deepEqual(a, b) {
  if (Object.is(a, b)) return true;
  if (!a || typeof a !== 'object' || !b || typeof b !== 'object') return false;
  if (a.constructor !== b.constructor) return false;
  if (a instanceof Date) return a.getTime() === b.getTime();
  if (a instanceof RegExp) return a.source === b.source && a.flags === b.flags;
  const kA = Reflect.ownKeys(a), kB = Reflect.ownKeys(b);
  if (kA.length !== kB.length) return false;
  for (const k of kA) {
    if (!Object.prototype.hasOwnProperty.call(b, k) || !deepEqual(a[k], b[k])) return false;
  }
  return true;
}
